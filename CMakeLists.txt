cmake_minimum_required(VERSION 3.16)

# Always read .duneos_board — cmake args (-DDUNEOS_BOARD=) are intentionally
# ignored so the VS Code extension cannot silently override the per-developer
# board choice stored in .duneos_board (which is gitignored).
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/.duneos_board")
    file(READ "${CMAKE_CURRENT_LIST_DIR}/.duneos_board" DUNEOS_BOARD)
    string(STRIP "${DUNEOS_BOARD}" DUNEOS_BOARD)
    # Make CMake re-run configure whenever .duneos_board changes.
    configure_file(
        "${CMAKE_CURRENT_LIST_DIR}/.duneos_board"
        "${CMAKE_CURRENT_BINARY_DIR}/.duneos_board.stamp"
        COPYONLY
    )
else()
    set(DUNEOS_BOARD "esp32s3-devkitc")
endif()

# Detect board switch without Full Clean — alert the user before sdkconfig
# drifts silently out of sync with the new board's settings.
set(_BOARD_STAMP "${CMAKE_CURRENT_BINARY_DIR}/duneos_last_board.stamp")
if(EXISTS "${_BOARD_STAMP}")
    file(READ "${_BOARD_STAMP}" _LAST_BOARD)
    string(STRIP "${_LAST_BOARD}" _LAST_BOARD)
    if(NOT "${_LAST_BOARD}" STREQUAL "${DUNEOS_BOARD}")
        message(FATAL_ERROR
            "\nBoard changed: '${_LAST_BOARD}' → '${DUNEOS_BOARD}'.\n"
            "The sdkconfig is stale and must be regenerated. Please do a Full Clean.\n"
            "  VS Code : Ctrl+Shift+P → ESP-IDF: Full Clean\n"
            "  CLI     : idf.py fullclean\n"
        )
    endif()
endif()
file(WRITE "${_BOARD_STAMP}" "${DUNEOS_BOARD}")

# SDKCONFIG_DEFAULTS must be set before include(project.cmake).
# Common settings first, board-specific settings override.
set(SDKCONFIG_DEFAULTS
    "sdkconfig.defaults"
    "boards/${DUNEOS_BOARD}/sdkconfig.defaults"
)

# IDF_TARGET from per-board idf_target.txt (generated by duneos-bspgen).
# This must be set before include(project.cmake).
set(_BOARD_TARGET_FILE "${CMAKE_CURRENT_LIST_DIR}/boards/${DUNEOS_BOARD}/idf_target.txt")
if(EXISTS "${_BOARD_TARGET_FILE}")
    file(READ "${_BOARD_TARGET_FILE}" _BOARD_IDF_TARGET)
    string(STRIP "${_BOARD_IDF_TARGET}" _BOARD_IDF_TARGET)
    set(IDF_TARGET "${_BOARD_IDF_TARGET}")
elseif(NOT DEFINED IDF_TARGET)
    set(IDF_TARGET esp32s3)
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(duneos)
